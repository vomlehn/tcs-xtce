# Each case: a time tool's input | the output it must give.
# Decoding: P-field 0x1e is CUC, CCSDS epoch, 4 coarse and 2 fine octets
cuc_decode 1e000000018000 | 7 3ff8000000000000
# P-field 0x2e: the agency's epoch; 1000 and a half seconds
cuc_decode 2e000003e88000 | 7 408f440000000000
# Extended P-field 9f 48: 4+2 coarse and 3+2 fine octets; 1 + 2^-40
cuc_decode 9f480000000000010000000001 | 13 3ff0000000001000
# A time code ID of 7 is not CUC
cuc_decode 7e000000018000 | none
# CDS P-field 0x40: CCSDS epoch, 16-bit days, no sub-ms; day 2 and the
# 86400th second of it, a leap second
cds_decode 40000205265c00 | 7 410fa40000000000
# 0x41: microseconds; day 2, 3004 ms and 5 us
cds_decode 41000200000bbc0005 | 9 410518180833c600
# A sub-millisecond code of 3 is reserved
cds_decode 43000000000000 | none
# Encoding: 1.999 s in 1+1 octets rounds 0.999*256 up to a carry
cuc_encode 1 1 0 3ffffbe76c8b4396 0 8 | 0200
# 255.999 s carries past what 1 coarse octet holds
cuc_encode 1 1 0 406ffff7ced91687 0 8 | none
# 2^32 s needs a fifth coarse octet
cuc_encode 4 0 0 41f0000000000000 0 8 | none
# Negative times have no code
cuc_encode 4 0 0 bff0000000000000 0 8 | none
# With its P-field; too little room says how much it needs
cuc_encode 4 2 1 408f440000000000 1 32 | 2e000003e88000
cuc_encode 4 2 1 408f440000000000 1 3 | needs 7
cuc_encode 6 5 0 0000000000000000 1 32 | 9f480000000000000000000000
# CDS: day 2, 3.004005 s, to the nearest microsecond
cds_encode 0 1 0 410518180833c600 1 32 | 41000200000bbc0005
# 0.9999996 s rounds to a whole second (1000 ms, 0 us)
cds_encode 0 1 0 3fefffff29406b2a 0 32 | 0000000003e80000
# The last millisecond of day 65535 rounds into day 65536: too late
cds_encode 0 0 0 41f517ffffffff97 0 32 | none
# Exact forms.  CUC 0x1e, 1.5 s: seconds 1, fraction 0x80 then nine noughts
cuc_decode_x 1e000000018000 | 7 0000000000000001 80000000000000000000
# Seven coarse and ten fine octets keep every bit: P-field 9f 7c, then the
# seven octets of seconds and the ten of fraction
cuc_encode_x 7 10 0 00ffeeddccbbaa99 88776655443322110001 1 32 | 9f7cffeeddccbbaa9988776655443322110001
# One fine octet: just under half a unit rounds down, half rounds up, and a
# fraction rounding up to a second carries; past the coarse octet, none
cuc_encode_x 1 1 1 0000000000000005 127fffffffffffffffff 0 2 | 0512
cuc_encode_x 1 1 1 0000000000000005 12800000000000000000 0 2 | 0513
cuc_encode_x 1 1 1 0000000000000005 ffffffffffffffffffff 0 2 | 0600
cuc_encode_x 1 1 1 00000000000000ff ffffffffffffffffffff 0 2 | none
# CDS 0x41: day 2, 3004 ms and 5 us, which is 5 000 000 ps
cds_decode_x 41000200000bbc0005 | 9 2 3004 5000000
# The last half millisecond of an ordinary day ends it; in a leap second
# the day goes on a second longer
cds_encode_x 0 0 0 9 86399999 500000000 0 6 | 000a00000000
cds_encode_x 0 0 0 9 86400000 500000000 0 6 | 000905265c01
cds_encode_x 0 0 0 9 86400999 500000000 0 6 | 000a00000000
# P-field 4e: agency epoch, 24-bit days, picoseconds; every picosecond kept
cds_encode_x 1 2 1 1193046 86399999 999999999 1 32 | 4e12345605265bff3b9ac9ff
# A picosecond field of 10^9 is out of range
cds_encode_x 0 2 0 0 0 1000000000 0 32 | none
